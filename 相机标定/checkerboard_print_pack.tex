\newcommand{\checkerboard}[3]{%
  \begin{tikzpicture}[x=1mm,y=1mm]
    \pgfmathtruncatemacro{\lastx}{#1-1}
    \pgfmathtruncatemacro{\lasty}{#2-1}
    \pgfmathsetmacro{\w}{#1*#3}
    \pgfmathsetmacro{\h}{#2*#3}
    \fill[white] (0,0) rectangle (\w,\h);
    \foreach \x in {0,...,\lastx} {
      \foreach \y in {0,...,\lasty} {
        \pgfmathtruncatemacro{\parity}{mod(\x+\y,2)}
        \ifnum\parity=0
          \fill[black] (\x*#3,\y*#3) rectangle ++(#3,#3);
        \fi
      }
    }
  \end{tikzpicture}%
}

\newcommand{\boardpage}[6]{%
  \begin{center}
    \vspace*{20mm}
    {\Large #1\par}
    \vspace{8mm}
    \checkerboard{#2}{#3}{#4}
    \vfill
    \begin{tabular}{>{\bfseries}r l}
      Printed squares: & #2 x #3 \\
      Board size: & #5 mm x #6 mm \\
      Square size: & #4 mm \\
      Print setting: & Actual size / 100\%, no fit-to-page scaling \\
    \end{tabular}
    \vspace*{12mm}
  \end{center}
  \clearpage
}
